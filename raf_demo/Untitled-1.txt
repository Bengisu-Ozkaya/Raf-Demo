import 'package:flutter/material.dart';
import 'package:provider/provider.dart';

import '../providers/cart_provider.dart';
import '../providers/auth_provider.dart';

class CartScreen extends StatelessWidget {
  static const routeName = '/cart';

  const CartScreen({super.key});

  @override
  Widget build(BuildContext context) {
    final cart = Provider.of<CartProvider>(context);
    final cartShops = cart.shops.values.toList();

    return Scaffold(
      appBar: AppBar(
        title: const Text('Sepetim'),
      ),
      body: cartShops.isEmpty
          ? const Center(
              child: Text('Sepetiniz boş.', style: TextStyle(fontSize: 20)),
            )
          : Column(
              children: <Widget>[
                Expanded(
                  child: ListView.builder(
                    itemCount: cartShops.length,
                    itemBuilder: (ctx, i) {
                      final shop = cartShops[i];
                      final items = shop.items.values.toList();
                      return Card(
                        margin: const EdgeInsets.symmetric(horizontal: 15, vertical: 8),
                        child: Padding(
                          padding: const EdgeInsets.all(8.0),
                          child: Column(
                            crossAxisAlignment: CrossAxisAlignment.start,
                            children: [
                              // Dükkan Adı
                              Padding(
                                padding: const EdgeInsets.all(8.0),
                                child: Text(
                                  shop.shopName,
                                  style: const TextStyle(fontSize: 20, fontWeight: FontWeight.bold),
                                ),
                              ),
                              const Divider(),
                              // O dükkana ait ürünler
                              ListView.builder(
                                shrinkWrap: true,
                                physics: const NeverScrollableScrollPhysics(),
                                itemCount: items.length,
                                itemBuilder: (ctx, j) => CartListItem(
                                  item: items[j],
                                  shopId: shop.shopId, // shopId'yi buraya ekle
                                ),
                              ),
                            ],
                          ),
                        ),
                      );
                    },
                  ),
                ),
                // Toplam Tutar ve Sipariş Butonu
                Card(
                  elevation: 6,
                  margin: const EdgeInsets.all(15),
                  child: Padding(
                    padding: const EdgeInsets.all(8),
                    child: Row(
                      mainAxisAlignment: MainAxisAlignment.spaceBetween,
                      children: <Widget>[
                        const Text(
                          'Toplam:',
                          style: TextStyle(fontSize: 20, fontWeight: FontWeight.bold),
                        ),
                        const Spacer(),
                        Chip(
                          label: Text(
                            '${cart.totalAmount.toStringAsFixed(2)} TL',
                            style: const TextStyle(
                              color: Colors.white,
                              fontWeight: FontWeight.bold,
                            ),
                          ),
                          backgroundColor: Theme.of(context).primaryColor,
                        ),
                        const SizedBox(width: 10),
                        OrderButton(cart: cart),
                      ],
                    ),
                  ),
                )
              ],
            ),
    );
  }
}

// Sipariş butonunu state management için ayrı bir widget'a alalım
class OrderButton extends StatefulWidget {
  const OrderButton({
    super.key,
    required this.cart,
  });

  final CartProvider cart;

  @override
  State<OrderButton> createState() => _OrderButtonState();
}

class _OrderButtonState extends State<OrderButton> {
  bool _isLoading = false;

  @override
  Widget build(BuildContext context) {
    return TextButton(
      onPressed: (widget.cart.totalAmount <= 0 || _isLoading)
          ? null
          : () async {
              setState(() { _isLoading = true; });
              final userId = Provider.of<AuthProvider>(context, listen: false).user!.id;
              try {
                await widget.cart.placeOrder(userId);
                ScaffoldMessenger.of(context).showSnackBar(
                  const SnackBar(content: Text('Siparişiniz başarıyla alındı!')),
                );
              } catch (error) {
                 ScaffoldMessenger.of(context).showSnackBar(
                  SnackBar(content: Text('Sipariş verilemedi: ${error.toString()}')),
                );
              }
              if (mounted) {
                setState(() { _isLoading = false; });
              }
            },
      child: _isLoading ? const CircularProgressIndicator() : const Text('SİPARİŞİ TAMAMLA'),
    );
  }
}

// Sepetteki her bir satırı temsil eden widget
class CartListItem extends StatelessWidget {
  final CartItem item;
  final int shopId; // shopId'yi burada al

  const CartListItem({super.key, required this.item, required this.shopId});

  @override
  Widget build(BuildContext context) {
    final cart = Provider.of<CartProvider>(context, listen: false);

    // Stok bilgisini almak için ShopProvider'a erişim.
    // Bu, anlık stok kontrolü için önemlidir.
    final productStock = context.select<ShopProvider, int>((shopProvider) {
      final product = shopProvider.products.firstWhere(
        (p) => p.id == item.id,
        orElse: () => Product(id: item.id, name: item.name, price: item.price, stock: 0, shopId: shopId),
      );
      return product.stock;
    });

    return ListTile(
      leading: CircleAvatar(
        backgroundImage: NetworkImage(item.imageUrl ?? 'https://via.placeholder.com/150'),
        onBackgroundImageError: (_, __) {}, // Hata durumunda boş göster
      ),
      title: Text(item.name),
      subtitle: Text('Toplam: ${(item.price * item.quantity).toStringAsFixed(2)} TL'),
      trailing: Row(
        mainAxisSize: MainAxisSize.min,
        children: [
          IconButton(
            icon: const Icon(Icons.remove_circle_outline),
            onPressed: () {
               cart.decrementItem(item.id, shopId);
            },
          ),
          Text('${item.quantity} x'),
          IconButton(
            icon: const Icon(Icons.add_circle_outline),
            // Stok kontrolünü burada da yapalım
            onPressed: item.quantity < productStock
              ? () {
                  cart.incrementItem(item.id, shopId, stockLimit: productStock);
                }
              : null, // Butonu disable et
          ),
        ],
      ),
    );
  }
}
